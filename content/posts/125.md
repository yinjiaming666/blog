# flutter 路由监听

> 在Flutter中主要是用 `RouteObserver` 对实现了RouteAware接口的Widget进行监测，`RouteObserver` 需要在 `MaterialApp`
> 进行初始化，而 `RouteAware` 则需要在子页面中进行监听。

## 声明单例 `RouteObserver`

```dart
class AppRouteObserver {
  static final RouteObserver<ModalRoute<void>> _routeObserver = RouteObserver<ModalRoute<void>>();

  //这是个单例
  static final AppRouteObserver _appRouteObserver = AppRouteObserver._internal();

  AppRouteObserver._internal();

  RouteObserver<ModalRoute<void>> get routeObserver {
    return _routeObserver;
  }

  factory AppRouteObserver() {
    return _appRouteObserver;
  }
}
```

## 在 `MaterialApp` 监听 `RouteObserver`

```dart

MaterialApp(
  home: MyPage(),
  navigatorObservers: [AppRouteObserver().routeObserver],
);

```


### 在组件中监听 `RouteAware`
```dart
import 'package:flutter/material.dart';

class MyPage extends StatefulWidget {
  @override
  _MyPageState createState() => _MyPageState();
}

class _MyPageState extends State<MyPage> with RouteAware {
  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    // 注册 RouteAware 监听
    routeObserver.subscribe(this, ModalRoute.of(context)!);
  }

  @override
  void dispose() {
    // 取消注册 RouteAware 监听
    routeObserver.unsubscribe(this);
    super.dispose();
  }

  @override
  void didPush() {
    print("进入页面");
  }

  @override
  void didPushNext() {
    print("从当前页面push到另一个页面");
  }

  @override
  void didPop() {
    print("当前的页面被pop");
  }

  @override
  void didPopNext() {
    print("从上一页返回");
  }
}
```
