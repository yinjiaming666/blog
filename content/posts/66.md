# WaitGroup

> 可以看作是一个计数器，会阻塞主线程。可以使主线程等待说有协程结束。add 几个就要 done 几个

`w := sync.WaitGroup{} // 声明`

`w.Add(1) // 加一`

`w.Wait() // 阻塞主线程`

`w.Done() // 减一`

```go
package main

import (
   "fmt"
   "sync"
)

func main() {
   w := sync.WaitGroup{} // 声明计数器
   var a []int
   defer get(&a)
   for i := 0; i < 200; i++ {
      w.Add(1) // 计数器加一
      go worker(&w, i, &a)
   }
   w.Wait() // 等待
   fmt.Println("结束")
}

func worker(w *sync.WaitGroup, i int, arr *[]int) {
   *arr = append(*arr, i)
   defer w.Done() // 计数器减一
}

func get(i *[]int) {
   fmt.Println(*i)
}
```
