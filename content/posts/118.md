# docker 部署 mysql 和 redis

```shell

docker run -p 63791:6379 --name redis --restart always -d -v /data/redis/redis.conf:/etc/redis/redis.conf -v /data/redis/data:/data  redis /etc/redis/redis.conf

docker run -p 33061:3306 --name mysql --restart always -d -v /data/mysql/conf:/etc/mysql/conf.d -v /data/mysql/logs:/logs -v /data/mysql/data:/var/lib/mysql -e MYSQL_ROOT_PASSWORD=123456 -d mysql:5.7

# mariadb
# mariadb 10.3 以上对应 mysql5.7
# mariadb 10.3 以上对应 mysql8
docker run --name mariadb11 --restart always -p 33063:3306 -e MYSQL_ROOT_PASSWORD=123546 -v /data/mariadb11/data:/var/lib/mysql -d mariadb:11.8.5-ubi9
```
