<script>
    const form = document.querySelector('#form')
    const img = document.querySelector('#img')
    const ipt = document.querySelector('#iptfile')
    const divPreview = document.querySelector("#preview")
    let file = null;
    let fileName = (Date.now() + rand(1000, 9999)).toString();
    let ext = '';

    function rand(min, max) {
        return Math.floor(Math.random() * (max - min)) + min;
    }

    let sliceNum = 0; // 切片数量
    let hasUploadNum = 0; // 已经上传切片数量

    /**
     * @returns {File[]}
     * @param file
     */
    function sliceFile(file) {
        // 切成多大一份
        let uploadSize = 1024 * 1024
        let arr = []
        for (let i = 0; i < file.size; i += uploadSize) {
            const index = (i / uploadSize) + 1
            // 文件切片  截取返回的是一个二进制文件
            arr.push(file.slice(i, uploadSize * index))
        }
        sliceNum = arr.length;
        return arr;
    }

    // 预览功能 隐藏input元素
    ipt.style.opacity = 0;
    ipt.addEventListener('change', (e) => {
        ext = '.' + e.target.files[0].name.split('.').pop().toLowerCase();
        file = new File([e.target.files[0]], fileName);
        divPreview.innerText = `FileName: ${file.name}, size: ${getSize(file)}`
    })

    // 格式化获取文件大小的函数
    function getSize(file) {
        // 传入文件对象files，一般就是FileList对象按索引来取一个对象
        // files.size读取文件大小
        const aMultiples = ["KiB", "MiB", "GiB", "TiB", "PiB", "EiB", "ZiB", "YiB"];
        let fSize = file.size
        let outSize = fSize + 'bytes'
        let nMult;
        for (fSize /= 1024, nMult = 0; fSize > 1; fSize /= 1024, nMult++) {
            outSize = fSize.toFixed(3) + aMultiples[nMult]
        }
        return outSize
    }

    // 发起上传文件的post请求
    function sendFile(file, sort) {
        let xhr = new XMLHttpRequest();
        let fd = new FormData();
        fd.append('file', file);
        fd.append('sort', sort);
        fd.append('num', sliceNum);
        xhr.open('POST', 'http://app.ycfhtc.com:9080/server/login/uploadSliceFile', true)
        xhr.onreadystatechange = function () {
            if (xhr.readyState === 4 && xhr.status === 200) {
                // 上传成功
                hasUploadNum = hasUploadNum + 1;
                let divPro = document.querySelector('#progress')
                divPro.style.width = hasUploadNum / sliceNum * 100 + '%'
                divPro.innerText = hasUploadNum / sliceNum * 100 + '%'
                console.log('请求成功, 数据为：', xhr.responseText);
                // if (hasUploadNum === sliceNum) {
                //     // 调用合并文件接口
                //     let xhr = new XMLHttpRequest();

                //     let fd = new FormData();
                //     fd.append('name', fileName);
                //     xhr.send(fd);
                // }
            }
        }
        xhr.send(fd)
    }

    form.addEventListener('submit', async function (e) {
        e.preventDefault();
        let arr = sliceFile(file);
        for (let i = 0; i < arr.length; i++) {
            let f = new File([arr[i]], '2024-01-10/' + fileName + ext);
            await sendFile(f, i + 1);
        }
    });

后端示例

/**
 * 分片上传文件
 * @param Request $request
 * @param $response
 * @param $redisCli
 * @param $mysql
 * @return array
 * @throws Exception
 * @author yin
 * @ApiTitle  (分片上传文件)
 * @ApiMethod (POST)
 * @ApiRoute  ({{url}}/server/login/uploadSliceFile)
 * @ApiParams (key=sort, value=1, type=text, nullable=false, description=当前是第几个)
 * @ApiParams (key=num, value=1, type=text, nullable=false, description=一共几个)
 * @ApiParams (key=file, value=, type=file, nullable=false, description=命名规则（2024-01-10/1114441555999.mp4）)
 * @ApiQuery  (key=jwt, value={{jwt}}, type=text, nullable=false, description=)
 */
public static function uploadSliceFile(Request $request, $response, $redisCli, $mysql): array
{
    $sort = $request->postInt('sort'); // 当前是第几个
    $num = $request->postInt('num'); // 一共几个

    $rootPath = dirname(__FILE__, 3) . '/nginx/static/slice_temp/';
    $file = $request->getRequest()->files['file'];
    $fileTmpPath = $file['tmp_name'];

    $file['name'] = explode('/', $file['name']);
    $file['name'] = end($file['name']); // 1114441555999.mp4
    var_dump($file['name']);
    $ext = explode('.', $file['name']);
    $ext = end($ext); // 文件类型 mp4

    $name = explode('.', $file['name']);
    $name = $name[0]; // 1114441555999
    $folderName = $rootPath . $name . DIRECTORY_SEPARATOR; // /nginx/static/slice_temp/1114441555999/
    if (!file_exists($folderName)) {
        mkdir($folderName, 0777, true);
    }
    $localFileName = $folderName . "$sort.$ext"; // /nginx/static/slice_temp/1114441555999/1.mp4
    if (file_exists($localFileName)) {
        return Utils::responseFail([], 10001, '重复上传的切片');
    }
    $res = move_uploaded_file($fileTmpPath, $localFileName);

    if (!$res) {
        return Utils::responseFail([], 10001, '文件上传失败');
    }

    Timer::after(3600 * 1000, function () use ($folderName) {
        // 超时释放资源
        Utils::delDir($folderName);
    });

    if ($num === count(scandir($folderName)) - 2) {
        // 合并文件
        $readAllFile = function (string $folder) {
            $handler = opendir($folder);
            $arr = [];
            while (($filename = readdir($handler)) !== false) {
                if ($filename != '.' && $filename != '..' && $filename !== '') {
                    $arr[] = $filename;
                }
            }
            closedir($handler);
            // 给文件排序
            for ($i = 0; $i < count($arr); $i++) {
                for ($j = $i + 1; $j < count($arr); $j++) {
                    if (intval($arr[$i]) > intval($arr[$j])) {
                        $tmp = $arr[$i]; // 这里的tmp是临时变量
                        $arr[$i] = $arr[$j]; // 第一次更换位置
                        $arr[$j] = $tmp; // 完成位置互换
                    }
                }
            }
            return $arr;
        };
        $fileList = $readAllFile($folderName);
        if (count($fileList) <= 0) {
            return Utils::responseFail(['folder_name' => $folderName], 10001, '未查询到文件，请重新上传');
        }

        $writeFile = dirname(__FILE__, 3) . '/nginx/static/' . date('Y-m-d') . "/$name.$ext"; // 要写入的文件
        $wf = fopen($writeFile, 'wb');
        foreach ($fileList as $fileName) {
            $fullFilePath = $folderName . $fileName;
            $handle = fopen($fullFilePath, 'rb');
            fwrite($wf, fread($handle, filesize($fullFilePath)));
            fclose($handle);
            unset($handle);
        }
        fclose($wf);
        unset($wf);

        go(function () use ($folderName) {
            Utils::delDir($folderName);
        });

        $url = 'static/' . date('Y-m-d') . "/$name.$ext"; // 外网访问路径
        if ($ext === 'mp4') {
            $toFile = substr($writeFile, 0, -4) . '.jpg'; // 要保存的内部路径
            $firstImgPath = substr($url, 0, -4) . '.jpg'; // 外网访问路径
            go(function () use ($writeFile, $toFile) {
                $str = "ffmpeg -i $writeFile -ss 00:00:01 -vframes 1 $toFile"; // 从第 1 秒开始，截取 1 帧
                var_dump($str);
                exec($str, $output, $returnVal);
                if ($returnVal !== 0) {
                    var_dump('视频第一帧截取失败' . $returnVal . $toFile);
                } else {
                    var_dump('视频第一帧截取成功：' . $toFile);
                }
            });
        }
        return Utils::responseSuccess([
            'url' => $url,
            'full_url' => QYSConfig::get('url') . '/' . $url,
            'first_img_path' => $firstImgPath ?? ''
        ]);
    }
    return Utils::responseSuccess([
        'url' => '',
        'full_url' => '',
        'first_img_path' => ''
    ]);
}
