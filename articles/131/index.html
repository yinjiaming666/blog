Golang redis 分布式锁

2026-10-22 · Go

通过 Redis Lua 脚本保证原子性，使用 EXISTS 检查锁、PSETEX 设置带 TTL 的锁，每 10 秒通过 GET 校验 Token 并使用 PEXPIRE 续约 30 秒，续约失败则取消业务 Context，执行结束后通过 GET 校验所有权并使用 DEL 安全释放锁，避免误删其他请求的锁。

代码

// Package redislock coordinates cooperating writers. SQL transactions, unique
// constraints and conditional updates remain the authority for persisted data.
package redislock

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"errors"
	"fmt"
	"sort"
	"strings"
	"time"

	redis "github.com/redis/go-redis/v9"
)

var (
	ErrBusy      = errors.New("redis resources remain locked")
	ErrLostLease = errors.New("redis lock lease was lost")
)

const acquireScript = `
for _, key in ipairs(KEYS) do
 if redis.call('EXISTS', key) ~= 0 then return 0 end
end
for _, key in ipairs(KEYS) do redis.call('PSETEX', key, ARGV[2], ARGV[1]) end
return 1`
const renewScript = `
for _, key in ipairs(KEYS) do
 if redis.call('GET', key) ~= ARGV[1] then return 0 end
end
for _, key in ipairs(KEYS) do redis.call('PEXPIRE', key, ARGV[2]) end
return 1`
const checkScript = `
for _, key in ipairs(KEYS) do
 if redis.call('GET', key) ~= ARGV[1] then return 0 end
end
return 1`
const releaseScript = `
for _, key in ipairs(KEYS) do
 if redis.call('GET', key) == ARGV[1] then redis.call('DEL', key) end
end
return 1`

// Manager uses one Redis server and automatically renews a bounded lease.
// Resource names must describe every resource changed by the callback. They are
// sorted and acquired atomically, so callers cannot deadlock on reversed orders.
type Manager struct {
	client                                            *redis.Client
	prefix                                            string
	ttl, renewInterval, wait, retry, operationTimeout time.Duration
	totalTimeout                                      time.Duration
}

func New(client *redis.Client, prefix string) (*Manager, error) {
	if client == nil {
		return nil, errors.New("redis lock client is nil")
	}
	prefix = strings.TrimSpace(prefix)
	if prefix == "" {
		return nil, errors.New("redis lock prefix is empty")
	}
	// WithTimeout clones the options and shares the caller-owned connection pool.
	// Respect short operation contexts even when the original general-purpose
	// client ignores deadlines. One attempt avoids ambiguous EVAL retransmission.
	bounded := client.WithTimeout(2 * time.Second)
	bounded.Options().ContextTimeoutEnabled = true
	bounded.Options().MaxRetries = 0
	return &Manager{client: bounded, prefix: strings.TrimRight(prefix, ":") + ":", ttl: 30 * time.Second, renewInterval: 10 * time.Second, wait: 5 * time.Second, retry: 25 * time.Millisecond, operationTimeout: 2 * time.Second, totalTimeout: 30 * time.Second}, nil
}

type leaseKey struct{}
type lease struct {
	manager *Manager
	keys    []string
	token   string
	ctx     context.Context
	cancel  context.CancelCauseFunc
}

func (m *Manager) resourceKeys(resources []string) ([]string, error) {
	if m == nil || m.client == nil {
		return nil, errors.New("redis lock manager is nil")
	}
	unique := make(map[string]struct{}, len(resources))
	for _, resource := range resources {
		if strings.TrimSpace(resource) == "" {
			return nil, errors.New("redis lock resource is empty")
		}
		unique[m.prefix+resource] = struct{}{}
	}
	if len(unique) == 0 {
		return nil, errors.New("redis lock resources are empty")
	}
	keys := make([]string, 0, len(unique))
	for key := range unique {
		keys = append(keys, key)
	}
	sort.Strings(keys)
	return keys, nil
}

// With applies one total deadline to acquisition and the callback. An earlier
// parent deadline wins. Timeout cancels the callback and stops renewal; explicit
// release waits for the callback to return. A callback that ignores cancellation
// can outlive its expiring lease, so callbacks must honor ctx and call Check
// immediately before committing side effects. Redis alone cannot fence MySQL.
func (m *Manager) With(ctx context.Context, resources []string, fn func(context.Context) error) error {
	if m == nil {
		return errors.New("redis lock manager is nil")
	}
	if m.totalTimeout <= 0 {
		return errors.New("redis lock total timeout must be positive")
	}
	ctx, cancelTotal := context.WithTimeout(ctx, m.totalTimeout)
	defer cancelTotal()
	if fn == nil {
		return errors.New("redis lock callback is nil")
	}
	keys, err := m.resourceKeys(resources)
	if err != nil {
		return err
	}
	if err = ctx.Err(); err != nil {
		return err
	}
	random := make([]byte, 24)
	if _, err = rand.Read(random); err != nil {
		return fmt.Errorf("create Redis lock token: %w", err)
	}
	token := hex.EncodeToString(random)
	if err = m.acquire(ctx, keys, token); err != nil {
		return err
	}
	callbackContext, cancel := context.WithCancelCause(ctx)
	held := &lease{manager: m, keys: keys, token: token, ctx: callbackContext, cancel: cancel}
	callbackContext = context.WithValue(callbackContext, leaseKey{}, held)
	stop, done := make(chan struct{}), make(chan struct{})
	go held.renew(stop, done)
	// Cleanup also runs when a callback panics. Token comparison never releases a
	// successor's lease, even if this callback outlived its TTL.
	defer func() {
		close(stop)
		cancel(context.Canceled)
		<-done
		cleanup, cancelCleanup := context.WithTimeout(context.Background(), m.operationTimeout)
		defer cancelCleanup()
		_ = m.client.Eval(cleanup, releaseScript, keys, token).Err()
	}()
	if callbackContext.Err() != nil {
		return context.Cause(callbackContext)
	}
	err = fn(callbackContext)
	return errors.Join(err, Check(callbackContext))
}

func (m *Manager) acquire(ctx context.Context, keys []string, token string) error {
	waiting, cancel := context.WithTimeout(ctx, m.wait)
	defer cancel()
	for {
		if err := waiting.Err(); err != nil {
			if ctx.Err() != nil {
				return ctx.Err()
			}
			return fmt.Errorf("%w: %w", ErrBusy, err)
		}
		operation, cancelOperation := context.WithTimeout(waiting, m.operationTimeout)
		acquired, err := m.client.Eval(operation, acquireScript, keys, token, m.ttl.Milliseconds()).Int64()
		cancelOperation()
		if err != nil {
			// A lost reply may mean SET succeeded. Only this token may be cleaned up.
			cleanup, cancelCleanup := context.WithTimeout(context.Background(), m.operationTimeout)
			_ = m.client.Eval(cleanup, releaseScript, keys, token).Err()
			cancelCleanup()
			if ctx.Err() != nil {
				return ctx.Err()
			}
			if waiting.Err() != nil {
				return fmt.Errorf("%w: %w", ErrBusy, waiting.Err())
			}
			return fmt.Errorf("acquire Redis resources: %w", err)
		}
		if acquired == 1 {
			return nil
		}
		timer := time.NewTimer(m.retry)
		select {
		case <-waiting.Done():
			timer.Stop()
		case <-timer.C:
		}
	}
}

func (l *lease) renew(stop <-chan struct{}, done chan<- struct{}) {
	defer close(done)
	ticker := time.NewTicker(l.manager.renewInterval)
	defer ticker.Stop()
	for {
		select {
		case <-stop:
			return
		case <-l.ctx.Done():
			return
		case <-ticker.C:
			operation, cancel := context.WithTimeout(l.ctx, l.manager.operationTimeout)
			renewed, err := l.manager.client.Eval(operation, renewScript, l.keys, l.token, l.manager.ttl.Milliseconds()).Int64()
			cancel()
			if err != nil || renewed != 1 {
				l.cancel(errors.Join(ErrLostLease, err))
				return
			}
		}
	}
}

// Check verifies ownership using the callback context. It also works outside a
// lease, allowing transaction helpers to retain their ordinary context checks.
func Check(ctx context.Context) error {
	if err := ctx.Err(); err != nil {
		return context.Cause(ctx)
	}
	held, _ := ctx.Value(leaseKey{}).(*lease)
	if held == nil {
		return nil
	}
	operation, cancel := context.WithTimeout(ctx, held.manager.operationTimeout)
	owned, err := held.manager.client.Eval(operation, checkScript, held.keys, held.token).Int64()
	cancel()
	if err != nil || owned != 1 {
		failure := errors.Join(ErrLostLease, err)
		held.cancel(failure)
		return failure
	}
	if ctx.Err() != nil {
		return context.Cause(ctx)
	}
	return nil
}
