PHP8 注解的使用

2023-03-24 · PHP

php8 注解

参考地址

声明注解

<?php

/**
 * @param $argument
 * @return void
 * @author yin
 */
#[Route(name: '/api/posts/{id}', methods: ['GET'])]
#[FunctionName('dummy')]
function dummy($argument): void
{
}

$ref = new ReflectionFunction('dummy');
var_dump($ref->getAttributes('Route')[0]->getArguments()['name']);
var_dump($ref->getAttributes('FunctionName')[0]->getArguments());

输出

string(15) "/api/posts/{id}"
array(1) {
[0]=>
string(5) "dummy"
}

$refClass->getAttributes('Route')[0]->getArguments() 会返回一个数组

声明注解类

#[Attribute(Attribute::TARGET_FUNCTION | Attribute::TARGET_CLASS)]

#[Attribute(Attribute::TARGET_FUNCTION | Attribute::TARGET_CLASS)]
class Route
{
    public string $name;
    public array $methods;

    public function __construct(string $name, array $methods)
    {
        $this->name = $name;
        $this->methods = $methods;
        var_dump($name);
        var_dump($methods);
    }
}

使用注解类

/**
 * @param $argument
 * @return void
 * @author yin
 */
#[Route(name: '/api/posts/{id}', methods: ['GET'])]
function dummy($argument): void
{
}

$refClass = new ReflectionFunction('dummy');
$route = $refClass->getAttributes('Route')[0]->newInstance(); // 会调用 Route 类的构造函数，并返回实例对象
var_dump($route->name);


#[Route(name: '/api/posts/{id}', methods: ['GET'])]
const T = 'name'; // 会报错
