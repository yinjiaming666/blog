channel

2023-02-26 · Go

声明

make 一个 channel 有两种方式，分别是有缓冲的和没缓冲的
有缓冲的 channel c := make(chan string, 10) make(chan TYPE,SIZE)
无缓冲的 channel c := make(chan string) make(chan TYPE)

阻塞

channel 塞满了，没有取出，就会导致阻塞。会报 fatal error: all goroutines are asleep - deadlock! 

// 往三个长度的 channel 插入 4 个元素会导致阻塞
package main

func main() {
   c := make(chan int, 3)
   c <- 1
   c <- 2
   c <- 3
   c <- 4
}


// 通过协程实时从 channel 取出就不会阻塞了
package main

func main() {
   c := make(chan int, 3)
   go get(c)
   c <- 1
   c <- 2
   c <- 3
   c <- 4
}

func get(c chan int) {
   for {
      println(<-c)
   }
}

select case 用法

for select 如果没有 default 的话，channel 的元素取光了会报错 goroutine 1 [select (no cases)] 造成阻塞

package main

import (
   "fmt"
   "time"
)

func main() {
   chan1 := make(chan string)
   chan2 := make(chan string)
   go pop(chan1, chan2)
   go setChan1(chan1)
   go setChan2(chan2)
   for {
      time.Sleep(1)
   }
}

func pop(c1, c2 chan string) {
   for {
      select {
      case msg1 := <-c1:
         fmt.Println(msg1)
      case msg2 := <-c2:
         fmt.Println(msg2)
      default:
         fmt.Println("no case ok")
      }
   }
}

func setChan1(c chan string) {
   c <- "chan1写入 ----- "
   time.Sleep(1000)
}

func setChan2(c chan string) {
   c <- "chan2写入 ----- "
   time.Sleep(1000)
}
